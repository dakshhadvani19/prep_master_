PREPMASTER FEEDBACK — REMOVED RATING UI ARCHIVE
================================================

This is a plain-text archive of the former five-star feedback rating UI.
It is intentionally not imported by the application. It preserves the
previous rating state, markup, hover/focus animation and CSS for reference.

Former imports and state
------------------------

import { useState } from 'react';
import { Star } from 'lucide-react';

const RATING_LABELS = ['', 'Poor', 'Fair', 'Good', 'Great', 'Excellent'];

const [rating, setRating] = useState(0);
const [hoveredStar, setHoveredStar] = useState(0);

Former JSX
----------

<div className="feedback-rating">
    <label className="feedback-rating__label">
        Only Rating
        {rating > 0 && (
            <span className="feedback-rating__value">
                — {RATING_LABELS[rating]}
            </span>
        )}
    </label>
    <div className="feedback-rating__stars" role="group" aria-label="Only Rating">
        {[1, 2, 3, 4, 5].map((star) => (
            <button
                type="button"
                key={star}
                className="feedback-rating__star"
                aria-label={star + ' out of 5'}
                aria-pressed={rating === star}
                onMouseEnter={() => setHoveredStar(star)}
                onMouseLeave={() => setHoveredStar(0)}
                onFocus={() => setHoveredStar(star)}
                onBlur={() => setHoveredStar(0)}
                onClick={() => setRating(star)}
            >
                <Star
                    size={32}
                    fill={(hoveredStar || rating) >= star ? '#fbbf24' : 'transparent'}
                    color={(hoveredStar || rating) >= star ? '#fbbf24' : 'rgba(255,255,255,0.2)'}
                />
            </button>
        ))}
    </div>
</div>
